run the app from the project folder:

    python3 -m http.server 8000

then open http://localhost:8000.

first check that History can be empty. use a fresh browser profile, or press Reset Score and confirm it. open History and make sure it says there’s nothing there. Reset Score should still keep the work minutes like before.

to check all the row types without waiting hours or a whole day, open DevTools, go to Console, paste this and hit Enter. it puts sample entries in local storage then reloads the app. the dated entries are mixed up on purpose, and one old entry is missing the date and work time.

    const key = 'kartavya-main-data';
    const saved = JSON.parse(localStorage.getItem(key) || '{}');
    const ago = minutes => new Date(Date.now() - minutes * 60_000).toISOString();
    const pad = value => String(value).padStart(2, '0');
    const now = new Date();
    const dateKey = [now.getFullYear(), pad(now.getMonth() + 1), pad(now.getDate())].join('-');
    saved.tasks = [];
    saved.taskStore = [];
    saved.activeTaskId = null;
    saved.lastStarTickAt = Date.now();
    saved.history = [
      { id: 'legacy', taskName: 'Older saved task', result: 'success', reward: 50 },
      { id: 'successful-task', taskName: 'Morning study', result: 'success', completedAt: ago(90), workedMs: 1800000, reward: 50 },
      { id: 'hourly', taskName: '★ Task shortfall', result: 'fail', completedAt: ago(30), isStarHourPenalty: true, penalty: 30, penaltyDate: dateKey + '-08' },
      { id: 'failed-task', taskName: 'Read a chapter', result: 'fail', completedAt: ago(60), workedMs: 900000, penalty: 200 },
      { id: 'daily', taskName: 'Daily starred task incomplete: *Stretch', result: 'fail', completedAt: ago(15), isStarDailyPenalty: true, penalty: 200, workedMs: 600000, requiredMs: 1800000 },
      { id: 'distraction', taskName: 'Distraction', result: 'fail', completedAt: ago(45), isDistractionPenalty: true, penalty: 10 }
    ];
    saved.starHourPenaltyApplied = saved.starHourPenaltyApplied || {};
    for (let hour = 8; hour < 22; hour++) saved.starHourPenaltyApplied[dateKey + '-' + pad(hour)] = true;
    localStorage.setItem(key, JSON.stringify(saved));
    location.reload();
